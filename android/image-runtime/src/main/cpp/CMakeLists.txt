cmake_minimum_required(VERSION 3.22)
project(notebuddy_image LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS image_compile=2 image_link=1)
set(CMAKE_JOB_POOL_COMPILE image_compile)
set(CMAKE_JOB_POOL_LINK image_link)
if(NOT EXISTS "${SD_SOURCE}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set imageRuntimePath to the verified pinned stable-diffusion.cpp checkout")
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(NB_IMAGE_BACKEND "cpu" CACHE STRING "Explicit image backend (cpu or opencl); no automatic fallback")
if(NOT NB_IMAGE_BACKEND MATCHES "^(cpu|opencl)$")
    message(FATAL_ERROR "Unsupported image backend; use cpu or opencl")
endif()
add_library(notebuddy_image SHARED image_jni.cpp)
target_compile_definitions(notebuddy_image PRIVATE NB_IMAGE_BACKEND="${NB_IMAGE_BACKEND}")
target_include_directories(notebuddy_image PRIVATE "${SD_SOURCE}/include" "${SD_SOURCE}/ggml/include")
if(SD_BUILD_PATH)
    # Reuse the pinned NDK arm64 Release build; never use a host inference binary.
    set(IMAGE_ARCHIVES
        "${SD_BUILD_PATH}/libstable-diffusion.a"
        "${SD_BUILD_PATH}/ggml/src/libggml.a"
        "${SD_BUILD_PATH}/ggml/src/libggml-cpu.a"
        "${SD_BUILD_PATH}/ggml/src/libggml-base.a")
    if(NB_IMAGE_BACKEND STREQUAL "opencl")
        file(STRINGS "${SD_BUILD_PATH}/CMakeCache.txt" OPENCL_ENABLED_LINE REGEX "^SD_OPENCL:BOOL=ON$")
        if(NOT OPENCL_ENABLED_LINE)
            message(FATAL_ERROR "Selected static build is not an OpenCL build")
        endif()
        if(NOT EXISTS "${OPENCL_LINK_LIBRARY}")
            message(FATAL_ERROR "OpenCL needs the verified Android link-time library (not bundled in APK)")
        endif()
        list(APPEND IMAGE_ARCHIVES "${SD_BUILD_PATH}/ggml/src/ggml-opencl/libggml-opencl.a")
        if(NOT EXISTS "${OPENCL_INCLUDE_DIR}/CL/cl.h")
            message(FATAL_ERROR "Set OPENCL_INCLUDE_DIR to the hash-verified Khronos headers")
        endif()
        target_include_directories(notebuddy_image PRIVATE "${OPENCL_INCLUDE_DIR}")
        target_compile_definitions(notebuddy_image PRIVATE NB_OPENCL=1)
        target_link_libraries(notebuddy_image PRIVATE "${OPENCL_LINK_LIBRARY}")
    endif()
    foreach(archive IN LISTS IMAGE_ARCHIVES)
        if(NOT EXISTS "${archive}")
            message(FATAL_ERROR "Finish the pinned arm64 Release build first: ${archive}")
        endif()
    endforeach()
    target_link_libraries(notebuddy_image PRIVATE "-Wl,--start-group" ${IMAGE_ARCHIVES} "-Wl,--end-group" log m dl atomic)
else()
    if(NB_IMAGE_BACKEND STREQUAL "opencl")
        message(FATAL_ERROR "OpenCL preparation requires the separately verified arm64 static build")
    endif()
    add_subdirectory("${SD_SOURCE}" sd-runtime EXCLUDE_FROM_ALL)
    target_link_libraries(notebuddy_image PRIVATE stable-diffusion log)
endif()
target_compile_features(notebuddy_image PRIVATE cxx_std_17)
