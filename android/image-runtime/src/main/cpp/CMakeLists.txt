cmake_minimum_required(VERSION 3.22)
project(notebuddy_image LANGUAGES C CXX)
set_property(GLOBAL PROPERTY JOB_POOLS image_compile=2 image_link=1)
set(CMAKE_JOB_POOL_COMPILE image_compile)
set(CMAKE_JOB_POOL_LINK image_link)
if(NOT EXISTS "${SD_SOURCE}/include/stable-diffusion.h")
    message(FATAL_ERROR "Set imageRuntimePath to the verified pinned stable-diffusion.cpp checkout")
endif()
set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_library(notebuddy_image SHARED image_jni.cpp)
target_include_directories(notebuddy_image PRIVATE "${SD_SOURCE}/include")
if(SD_BUILD_PATH)
    # Reuse the pinned NDK arm64 Release build; never use a host inference binary.
    set(IMAGE_ARCHIVES
        "${SD_BUILD_PATH}/libstable-diffusion.a"
        "${SD_BUILD_PATH}/ggml/src/libggml.a"
        "${SD_BUILD_PATH}/ggml/src/libggml-cpu.a"
        "${SD_BUILD_PATH}/ggml/src/libggml-base.a")
    foreach(archive IN LISTS IMAGE_ARCHIVES)
        if(NOT EXISTS "${archive}")
            message(FATAL_ERROR "Finish the pinned arm64 Release build first: ${archive}")
        endif()
    endforeach()
    target_link_libraries(notebuddy_image PRIVATE "-Wl,--start-group" ${IMAGE_ARCHIVES} "-Wl,--end-group" log m dl atomic)
else()
    add_subdirectory("${SD_SOURCE}" sd-runtime EXCLUDE_FROM_ALL)
    target_link_libraries(notebuddy_image PRIVATE stable-diffusion log)
endif()
target_compile_features(notebuddy_image PRIVATE cxx_std_17)
